            <p class="text-sm text-gray-500">拍照上傳，讓 AI 幫你秒速辨識與定價！</p>
        </header>

        <!-- 上傳區塊 -->
        <div class="bg-white p-6 rounded-2xl shadow-sm border border-gray-100 mb-6">
            <h2 class="font-bold text-lg mb-3">上架新商品</h2>
            <input type="file" id="imageInput" accept="image/*" class="mb-4 block w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100">
            <input type="text" id="sellerContact" placeholder="您的聯絡方式（如 LINE ID 或 Email）" class="w-full p-2 border rounded-lg mb-4 text-sm">
            <button onclick="uploadProduct()" id="submitBtn" class="w-full bg-blue-600 text-white font-bold py-2 rounded-xl hover:bg-blue-700 transition">開始 AI 智慧上架</button>
            <p id="loading" class="text-center text-sm text-blue-500 mt-2 hidden">AI 正在努力辨識商品中，請稍候...</p>
        </div>

        <!-- 商品列表 -->
        <h2 class="font-bold text-lg mb-3">熱門二手商品</h2>
        <div id="productList" class="space-y-4">
            <!-- 動態載入 -->
        </div>
    </div>

    <script>
        async function fetchProducts() {
            try {
                const res = await fetch('/api/products');
                const result = await res.json();
                if(result.success) {
                    const list = document.getElementById('productList');
                    list.innerHTML = result.data.map(p => `
                        <div class="bg-white p-4 rounded-xl shadow-sm border border-gray-100 flex justify-between items-center">
                            <div>
                                <h3 class="font-bold text-lg">${p.title}</h3>
                                <p class="text-xs text-gray-500">${p.description || ''}</p>
                                <p class="text-xs text-blue-600 mt-1">聯絡：${p.seller_contact}</p>
                            </div>
                            <div class="text-right">
                                <span class="text-lg font-bold text-red-500">NT$ ${p.price}</span>
                            </div>
                        </div>
                    `).join('');
                }
            } catch(e) { console.error(e); }
        }

        async function uploadProduct() {
            const fileInput = document.getElementById('imageInput');
            const contactInput = document.getElementById('sellerContact');
            const loading = document.getElementById('loading');
            const btn = document.getElementById('submitBtn');

            if(fileInput.files.length === 0) {
                alert('請先選擇一張商品照片！');
                return;
            }

            loading.classList.remove('hidden');
            btn.disabled = true;

            const reader = new FileReader();
            reader.readAsDataURL(fileInput.files[0]);
            reader.onload = async () => {
                try {
                    const res = await fetch('/api/create-product', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ 
                            imageBase64: reader.result,
                            sellerContact: contactInput.value 
                        })
                    });
                    const result = await res.json();
                    if(result.success) {
                        alert('上架成功！AI 判定名稱：' + result.aiResult.title);
                        fileInput.value = '';
                        contactInput.value = '';
                        fetchProducts();
                    } else {
                        alert('上架失敗：' + result.error);
                    }
                } catch(e) {
                    alert('發生網路錯誤');
                } finally {
                    loading.classList.add('hidden');
                    btn.disabled = false;
                }
            };
        }

        fetchProducts();
    </script>
